export type ArBlock =
  | { t: 'lead'; x: string }
  | { t: 'p'; x: string }
  | { t: 'h'; x: string }
  | { t: 'quote'; x: string; who?: string }
  | { t: 'img'; key: string; cap?: string }
  | { t: 'pull'; x: string };

export type Source = { label: string; url: string };

export type Article = {
  key: string;
  kicker: string;          // small line above the title
  title: string;
  standfirst: string;      // the deck under the title
  author?: string;         // who wrote it for ZAND, if original
  readMins: number;
  cover: string;
  hero?: string;           // optional portrait inside
  tag: 'ARCHITECTURE' | 'ART' | 'SCIENCE' | 'PROFILE' | 'ESSAY';
  blocks: ArBlock[];
  source?: Source;         // faded credit at the foot, links out
};
